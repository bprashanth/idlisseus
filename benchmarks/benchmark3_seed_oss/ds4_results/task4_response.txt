    <script>
        (function() {
            const canvas = document.getElementById('topicChart');
            if (!canvas) return;
            const ctx = canvas.getContext('2d');

            const dpr = window.devicePixelRatio || 1;
            const rect = canvas.getBoundingClientRect();
            const w = rect.width || 700;
            const h = rect.height || 380;
            canvas.width = w * dpr;
            canvas.height = h * dpr;
            canvas.style.width = w + 'px';
            canvas.style.height = h + 'px';
            ctx.scale(dpr, dpr);

            const topics = [
                'Human Activity',
                'Community\nParticipation',
                'Technology /\nRemote Sensing',
                'Ecological\nImpact',
                'Detection /\nEarly Warning',
                'Policy &\nGovernance',
                'Climate\nChange',
                'Post-fire\nRestoration'
            ];
            const counts = [5, 3, 3, 3, 3, 2, 2, 2];

            const colors = [
                '#f97316', '#fbbf24', '#f59e0b', '#ef7e1a',
                '#e5a83a', '#d98c2c', '#c9771e', '#b8651a'
            ];

            const barWidth = Math.min(56, (w - 80) / topics.length * 0.75);
            const gap = 18;
            const startX = 60;
            const bottomY = h - 40;
            const maxBarH = h - 100;
            const maxCount = 5;

            // background
            ctx.clearRect(0, 0, w, h);

            // subtle grid lines
            ctx.strokeStyle = '#3a4055';
            ctx.lineWidth = 0.6;
            for (let i = 0; i <= maxCount; i++) {
                const y = bottomY - (i / maxCount) * maxBarH;
                ctx.beginPath();
                ctx.moveTo(startX - 8, y);
                ctx.lineTo(w - 20, y);
                ctx.stroke();
                ctx.fillStyle = '#8892b0';
                ctx.font = '10px sans-serif';
                ctx.textAlign = 'right';
                ctx.textBaseline = 'middle';
                ctx.fillText(i, startX - 12, y);
            }

            // bars
            const totalWidth = topics.length * (barWidth + gap) - gap;
            const offsetX = startX + (w - startX - 20 - totalWidth) / 2;

            for (let i = 0; i < topics.length; i++) {
                const x = offsetX + i * (barWidth + gap);
                const barH = (counts[i] / maxCount) * maxBarH;
                const y = bottomY - barH;

                // bar shadow
                ctx.shadowColor = 'rgba(0,0,0,0.25)';
                ctx.shadowBlur = 6;
                ctx.shadowOffsetY = 2;

                // gradient bar
                const grad = ctx.createLinearGradient(0, y, 0, bottomY);
                grad.addColorStop(0, colors[i]);
                grad.addColorStop(1, colors[i] + 'cc');
                ctx.fillStyle = grad;
                ctx.beginPath();
                const radius = 4;
                const bX = x;
                const bW = barWidth;
                const bH = barH;
                const bY = y;
                // rounded rect
                ctx.moveTo(bX + radius, bY);
                ctx.lineTo(bX + bW - radius, bY);
                ctx.quadraticCurveTo(bX + bW, bY, bX + bW, bY + radius);
                ctx.lineTo(bX + bW, bY + bH - radius);
                ctx.quadraticCurveTo(bX + bW, bY + bH, bX + bW - radius, bY + bH);
                ctx.lineTo(bX + radius, bY + bH);
                ctx.quadraticCurveTo(bX, bY + bH, bX, bY + bH - radius);
                ctx.lineTo(bX, bY + radius);
                ctx.quadraticCurveTo(bX, bY, bX + radius, bY);
                ctx.fill();

                ctx.shadowColor = 'transparent';
                ctx.shadowBlur = 0;

                // count label on top
                ctx.fillStyle = '#e5e9f0';
                ctx.font = 'bold 11px sans-serif';
                ctx.textAlign = 'center';
                ctx.textBaseline = 'bottom';
                ctx.fillText(counts[i], x + barWidth / 2, y - 6);

                // topic label (multiline)
                const label = topics[i];
                const lines = label.split('\n');
                ctx.fillStyle = '#bcc1d4';
                ctx.font = '9px sans-serif';
                ctx.textAlign = 'center';
                ctx.textBaseline = 'top';
                const labelY = bottomY + 8;
                for (let li = 0; li < lines.length; li++) {
                    const lineY = labelY + li * 14;
                    ctx.fillText(lines[li], x + barWidth / 2, lineY);
                }
            }

            // subtle x-axis line
            ctx.strokeStyle = '#4c5267';
            ctx.lineWidth = 1;
            ctx.beginPath();
            ctx.moveTo(startX - 8, bottomY);
            ctx.lineTo(w - 20, bottomY);
            ctx.stroke();

            // title inside canvas
            ctx.fillStyle = '#9aa3bf';
            ctx.font = '9px sans-serif';
            ctx.textAlign = 'left';
            ctx.textBaseline = 'top';
            ctx.fillText('← papers →', startX + 20, h - 12);

        })();
    </script>
